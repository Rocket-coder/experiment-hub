run_id                                              —       id запуска
start_time                                          —       время старта по UTC
program                                             —       название запускаемой программы (файла)
parameters                                          —       параметры запуска
status                                              —       статус запуска (завершён, в процессе и т.д.)
result / output                                     —       результат запуска и возможный вывод
end_time                                            —       время завершения по UTC

execution_context:
    hostname                                        —       имя хоста
    os                                              —       операционная система
    python_version                                  —       версия Python (тут возможна и другая среда в дальнейшем)

optional:
    git_commit                                      —       коммит, из которого был запуск
    git_dirty                                       —       логический параметр, обозначающий не закомиченные изменения перед запуском
    dependency/enviroment info                      —       информация об зависимостях и/или настройках окружения
    traceback                                       —       трассировка
